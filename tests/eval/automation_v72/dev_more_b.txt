## device_numeric
auto: num(cover.kueche_rollladen).cover_position==0 => turn_on(light.kueche) :: Wenn der Küche Rollladen 0 Prozent erreicht, schalte das Küchenlicht ein.
auto: num(cover.schlafzimmer_rollladen).cover_position>50 => turn_off(fan.schlafzimmer) :: Wenn das Rollo im Schlafzimmer über 50 Prozent offen ist, schalte den Ventilator im Schlafzimmer aus.
auto: num(sensor.wohnzimmer_temperatur).state>25 => turn_on(fan.schlafzimmer) :: Sobald es im Wohnzimmer über 25 Grad hat, schalte den Schlafzimmer Ventilator ein.
auto: num(sensor.aussentemperatur).state<5 => turn_on(climate.wohnzimmer) :: Wenn die Außentemperatur unter 5 Grad fällt, schalte die Heizung im Wohnzimmer ein.
auto: num(sensor.buero_luftfeuchtigkeit).state<40 => turn_off(fan.schlafzimmer) :: Wenn die Luftfeuchtigkeit im Büro unter 40 Prozent fällt, mach den Ventilator im Schlafzimmer aus.
## device_time_sun
auto: time=06:00 => open(cover.schlafzimmer_rollladen) :: Um 6 Uhr öffne den Rollladen im Schlafzimmer.
auto: time=23:15 => turn_off(light.wohnzimmer) :: Um 23:15 Uhr mach das Wohnzimmerlicht aus.
auto: sun=sunset => close(cover.kueche_rollladen) :: Schließe bei Sonnenuntergang den Rollladen in der Küche.
auto: sun=sunrise => open(cover.bad_rollladen) :: Bei Sonnenaufgang fahre den Rollladen im Bad hoch.
auto: sun=sunset+10 => turn_on(light.flur) :: 10 Minuten nach Sonnenuntergang mach das Flurlicht an.

## device_conditions

auto: state(binary_sensor.kuechenfenster)=open ; if: state(climate.wohnzimmer)=on => turn_off(climate.wohnzimmer) :: Wenn das Küchenfenster aufgeht und die Wohnzimmer Heizung an ist, schalte die Wohnzimmer Heizung aus.
